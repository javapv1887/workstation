# Mac 프롬프트는 $, Ubuntu 프롬프트는 #으로 단순화. 입력 수정·화면 줄바꿈·커서 제어 문자와 Docker Debug 안내문은 제외.
# 컨테이너 목록은 ws-ubuntu 행을 발췌. attach 종료 후 docker ps는 실제로 헤더만 출력돼 그대로 보존.

$ docker start ws-ubuntu
ws-ubuntu

$ ps

45309 ttys000    0:00.01 script -a evidence/logs/attach-exec.txt

$ docker ps

8a1ebbb088f7   ubuntu:24.04   "bash"    22 hours ago   Up 10 seconds             ws-ubuntu

$ docker exec -it ws-ubuntu bash
# echo "exec test"
exec test
# exit
exit

$ docker ps
CONTAINER ID   IMAGE          COMMAND   CREATED        STATUS              PORTS     NAMES
8a1ebbb088f7   ubuntu:24.04   "bash"    22 hours ago   Up About a minute             ws-ubuntu

$ docker attach ws-ubuntu
# echo "attach test"
attach test
# exit
exit

$ docker ps
CONTAINER ID   IMAGE     COMMAND   CREATED   STATUS    PORTS     NAMES

$ docker ps -a
CONTAINER ID   IMAGE          COMMAND    CREATED          STATUS                      PORTS     NAMES
8a1ebbb088f7   ubuntu:24.04   "bash"     22 hours ago     Exited (0) 18 seconds ago             ws-ubuntu

$ exit
